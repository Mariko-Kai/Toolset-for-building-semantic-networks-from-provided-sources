\begin{theorem}[Uniform Convergence of a Series]
\[
\TerForall S \subseteq \RealNumbers
\TerForall u_k \colon S \TermTo \RealNumbers
\TerForall \Function{f} \colon S \TermTo \RealNumbers
\left( \TerForall x \in S \left( \sum_{k=1}^{\infty} u_k(x) = f(x) \right) \right)
\TermEquivalence
\left( \TerForall \varepsilon > 0 \TermExists N \in \NaturalNumbers \TerForall n > N \TerForall x \in S \left( \AbsAbstract{\mathrm{abs}}\left( f(x) - \sum_{k=1}^{n} u_k(x) \right) < \varepsilon \right) \right)
\]
\end{theorem}

\begin{proof}[RU]
Пусть $S$ — \Set{множество}, а $\sum u_k$ — ряд \Function{функций}, сходящийся поточечно к $\Function{f}$ на $S$. Согласно определению, равномерная сходимость означает, что для любого $\varepsilon > 0$ существует такое $N$, что для всех $n > N$ и всех $x \in S$ остаток ряда $\AbsAbstract{\mathrm{abs}}\left( f(x) - \sum_{k=1}^{n} u_k(x) \right)$ становится меньше $\varepsilon$. Это эквивалентно утверждению, что скорость сходимости не зависит от выбора точки $x \in S$.
\end{proof}

\begin{proof}[EN]
Let $S$ be a set and $\sum u_k$ be a series of functions converging pointwise to $\Function{f}$ on $S$. By definition, uniform convergence implies that for every $\varepsilon > 0$, there exists an integer $N$ such that for all $n > N$ and all $x \in S$, the remainder of the series $\AbsAbstract{\mathrm{abs}}\left( f(x) - \sum_{k=1}^{n} u_k(x) \right)$ is less than $\varepsilon$. This is equivalent to the statement that the rate of convergence is independent of the choice of the point $x \in S$.
\end{proof}
