\begin{operation}[Infimum]
\mForall{S \subseteq \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{L \in \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsInfimum}(L, S) \mDefinedAs
\left(
\mForall{s \in S} \quad L \leq s
\right)
\mAnd
\left(
\mForall{y \in \entityref{obj-real-numbers}{\mathbb{R}}}
\left(
\left(
\mForall{s \in S} \quad y \leq s
\right)
\mImplies
y \leq L
\right)
\right)
\end{operation}
