\begin{operation}[RiemannSum]
\mForall{n \colon \entityref{obj-natural-numbers}{\mathbb{N}}}
\mForall{a, b \colon \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{t \colon \{0, \dots, n\} \to \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{x \colon \{1, \dots, n\} \to \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{f \colon \entityref{obj-closed-interval}{[a, b]} \to \entityref{obj-real-numbers}{\mathbb{R}}}
\left( \entityref{prop-partition}{t, a, b, n} \mAnd \entityref{prop-tag-selection}{x, t, n} \right)
\mImplies
\mathrm{RiemannSum}(f, t, x, n) \mDefinedAs \sum_{i=1}^{n} f(x(i)) \cdot (t(i) - t(i-1))
\end{operation}
