\begin{operation}[Preimage]
\[
\mForall{X \colon \mathrm{Type}} \mForall{Y \colon \mathrm{Type}} \mForall{f \colon X \mTo Y} \mForall{B \subseteq Y}
\quad \mathrm{Preimage}(f, B) \mDefinedAs \{ x \in X \mid f(x) \in B \}
\]
\end{operation}
