\begin{object}[term-implication] % Импликация (Implication)

\textbf{Символ:} $\entityref{def-logical-connectives}{\implies}$ (также $\to$)

\textbf{Читается как:} «если..., то...», «влечет».

\textbf{Формальная семантика:}
Пусть $P$ и $Q$ — утверждения. Утверждение $P \entityref{def-logical-connectives}{\implies} Q$ ложно тогда и только тогда, когда $P$ истинно, а $Q$ ложно. В остальных случаях импликация истинна.

\textbf{Пример:} $(2 < 1) \entityref{def-logical-connectives}{\implies} (3 = 4)$
\end{object}
