\begin{object}[term-subset] % \entityref{obj-subset}{Подмножество} (Subset)



\textbf{Символ:} $\subset$ (или $\subseteq$)

\textbf{Формальное определение:}
\entityref{obj-set}{Множество} $A$ называется \entityref{obj-subset}{подмножеством} \entityref{obj-set}{множества} $B$, если каждый элемент $A$ является элементом $B$:
\[ A \subset B \mDefIff \mForall{x} (x \in A \mImplies x \in B) \]

\textbf{Примечание:} В архитектуре Mathesis символ $\subset$ является терминалом и не требует обертки в \texttt{entityref}.
\end{object}
