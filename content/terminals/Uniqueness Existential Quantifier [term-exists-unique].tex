\begin{object}[term-exists-unique] % Квантор существования и единственности (Uniqueness Existential Quantifier)

\textbf{Символ:} $\entityref{term-exists}{\exists}!$

\textbf{Читается как:} «существует единственный», «найдется один и только один».

\textbf{Формальная семантика:}
Пусть $P(x)$ — предикат. Утверждение $\entityref{term-exists}{\exists}! x\, P(x)$ истинно тогда и только тогда,
когда в области дискурса существует ровно один элемент $x$, для которого $P(x)$ истинно.

\textbf{Пример:} $\entityref{term-exists}{\exists}! x \in \entityref{obj-real-numbers}{\mathbb{R}}\; (x + 5 = 0)$
\end{object}
