\begin{object}[term-negation] % Отрицание (Negation)

\textbf{Символ:} $\lnot$

\textbf{Читается как:} «не», «неверно, что».

\textbf{Формальная семантика:}
Пусть $P$ — утверждение. Утверждение $\lnot P$ истинно тогда и только тогда, когда $P$ ложно.

\textbf{Пример:} $\lnot (2 = 3)$
\end{object}
