\begin{object}[term-mid] % Такое что (Such That)

\textbf{Символ:} $\entityref{op-mid}{\mid}$

\textbf{Читается как:} «такой, что», «при условии, что».

\textbf{Формальная семантика:}
В теории \entityref{obj-set}{множеств} ZFC символ $\entityref{op-mid}{\mid}$ (или $\colon$) используется в записи \entityref{obj-set}{множеств} методом выделения (Set-Builder Notation): $\{x \in A \entityref{op-mid}{\mid} P(x)\}$ обозначает \entityref{obj-set}{множество} всех элементов $x$ из $A$, для которых истинен предикат (свойство) $P(x)$.
Математически выражает отношение отбора или фильтрации элементов.

\textbf{Пример:} $\{x \in \entityref{obj-real-numbers}{\mathbb{R}} \entityref{op-mid}{\mid} x > 0\}$
\end{object}
