\begin{object}[term-forall] % Квантор всеобщности (Universal Quantifier)



\textbf{Символ:} $\forall$

\textbf{Читается как:} «для всех», «для каждого», «для любого».

\textbf{Формальная семантика:}
Пусть $P(x)$ — предикат. Утверждение $\forall x\, P(x)$ истинно тогда и только тогда,
когда $P(x)$ истинно для каждого элемента $x$ из области дискурса (универсума).

\textbf{Пример:} $\forall n \in \entityref{obj-natural-numbers}{\mathbb{N}}\; (n + 0 = n)$
\end{object}
