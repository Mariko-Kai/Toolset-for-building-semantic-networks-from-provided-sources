\begin{object}[term-in] % Принадлежность (Membership)



\textbf{Символ:} $\in$

\textbf{Читается как:} «принадлежит», «содержится в» (как элемент).

\textbf{Формальная семантика:}
В теории \entityref{obj-set}{множеств} ZFC символ $\in$ является неопределяемым базовым бинарным отношением.
Утверждение $x \in A$ означает, что объект $x$ является элементом \entityref{obj-set}{множества} $A$.

\textbf{Пример:} $1 \in \{1, 2, 3\}$
\end{object}
