\begin{object}[term-equivalence] % Эквивалентность (Equivalence)

\textbf{Символ:} $\entityref{def-logical-connectives}{\iff}$ (также $\entityref{op-leftrightarrow}{\leftrightarrow}$)

\textbf{Читается как:} «тогда и только тогда, когда», «эквивалентно».

\textbf{Формальная семантика:}
Пусть $P$ и $Q$ — утверждения. Утверждение $P \entityref{def-logical-connectives}{\iff} Q$ истинно тогда и только тогда, когда утверждения $P$ и $Q$ имеют одинаковые истинностные значения (оба истинны или оба ложны).

\textbf{Пример:} $(2 < 3) \entityref{def-logical-connectives}{\iff} (3 > 2)$
\end{object}
