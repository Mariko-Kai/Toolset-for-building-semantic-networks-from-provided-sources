\begin{theorem}[Uniform Convergence of a Series]
\[
\mForall{S \subseteq \entityref{obj-real-numbers}{\mReal}}
\mForall{u_k \colon S \mTo \entityref{obj-real-numbers}{\mReal}}
\mForall{\entityref{obj-function}{f} \colon S \mTo \entityref{obj-real-numbers}{\mReal}}
\left( \mForall{x \in S} \left( \entityref{op-sum}{\sum}_{k=1}^{\infty} u_k(x) = f(x) \right) \right)
\mIff
\left( \mForall{\varepsilon > 0} \mExists{N \in \entityref{obj-natural-numbers}{\mathbb{N}}} \mForall{n > N} \mForall{x \in S} \left( \entityref{op-abs-abstract}{\mathrm{abs}}\left( f(x) - \entityref{op-sum}{\sum}_{k=1}^{n} u_k(x) \right) < \varepsilon \right) \right)
\]
\end{theorem}

\begin{proof}[RU]
Пусть $S$ — \entityref{obj-set}{множество}, а $\entityref{op-sum}{\sum} u_k$ — ряд \entityref{obj-function}{функций}, сходящийся поточечно к $\entityref{obj-function}{f}$ на $S$. Согласно определению, равномерная сходимость означает, что для любого $\varepsilon > 0$ существует такое $N$, что для всех $n > N$ и всех $x \in S$ остаток ряда $\entityref{op-abs-abstract}{\mathrm{abs}}\left( f(x) - \entityref{op-sum}{\sum}_{k=1}^{n} u_k(x) \right)$ становится меньше $\varepsilon$. Это эквивалентно утверждению, что скорость сходимости не зависит от выбора точки $x \in S$.
\end{proof}

\begin{proof}[EN]
Let $S$ be a set and $\entityref{op-sum}{\sum} u_k$ be a series of functions converging pointwise to $\entityref{obj-function}{f}$ on $S$. By definition, uniform convergence implies that for every $\varepsilon > 0$, there exists an integer $N$ such that for all $n > N$ and all $x \in S$, the remainder of the series $\entityref{op-abs-abstract}{\mathrm{abs}}\left( f(x) - \entityref{op-sum}{\sum}_{k=1}^{n} u_k(x) \right)$ is less than $\varepsilon$. This is equivalent to the statement that the rate of convergence is independent of the choice of the point $x \in S$.
\end{proof}
