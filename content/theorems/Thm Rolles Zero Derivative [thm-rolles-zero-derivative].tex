\begin{theorem}[Rolle's Theorem on the Zero of a Derivative]
\mForall{\entityref{obj-function}{f} \colon \entityref{obj-closed-interval}{[0, 1]} \mTo \entityref{obj-real-numbers}{\mathbb{R}}}
\left( \entityref{prop-continuous}{f} \mAnd \mForall{a \in \entityref{obj-open-interval}{(0, 1)}} \entityref{op-derivative-at-point}{f'}(a) = 0 \right)
\mImplies f(0) = f(1)
\end{theorem}

\begin{proof}[RU]
Пусть \entityref{obj-function}{g}(x) = \entityref{obj-function}{f}(x) + cx. Так как \entityref{obj-derivative}{g'}(x) = \entityref{obj-derivative}{f'}(x) + c = c, то из \entityref{thm-mean-value-theorem}{теоремы о среднем значении} следует, что \entityref{obj-function}{f}(1) - \entityref{obj-function}{f}(0) > -c. Аналогично, рассматривая \entityref{obj-function}{h}(x) = cx - \entityref{obj-function}{f}(x), получаем \entityref{obj-function}{f}(1) - \entityref{obj-function}{f}(0) < c. Поскольку это верно для любого \entityref{obj-real-numbers}{c} > 0, заключаем, что \entityref{obj-function}{f}(0) = \entityref{obj-function}{f}(1).
\end{proof}

\end{proof}
