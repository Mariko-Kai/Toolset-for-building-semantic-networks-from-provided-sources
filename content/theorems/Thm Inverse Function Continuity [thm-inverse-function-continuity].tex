\begin{theorem}[Continuity of Inverse Function]
\[
\mForall{f \colon \entityref{obj-closed-interval}{[a, b]} \mTo \entityref{obj-closed-interval}{[c, d]}}
\mForall{g \colon \entityref{obj-closed-interval}{[c, d]} \mTo \entityref{obj-closed-interval}{[a, b]}}
\left( \entityref{prop-inverse-function}{g \circ f = \mathrm{id}_{[a,b]} \mAnd f \circ g = \mathrm{id}_{[c,d]}} \right)
\mImplies
\mForall{y_0 \in \entityref{obj-open-interval}{(c, d)}}
\left( \entityref{prop-continuous-at}{g, y_0} \right)
\]
\end{theorem}

\begin{proof}[RU]
Пусть $x_0 = \entityref{op-inverse-function}{g}(y_0)$, тогда $\entityref{op-function-application}{f}(x_0) = y_0$. Пусть задано $\varepsilon > 0$. Выберем $\delta = \entityref{op-minimum}{\min}(\entityref{op-function-application}{f}(x_0) - \entityref{op-function-application}{f}(x_0 - \varepsilon), \entityref{op-function-application}{f}(x_0 + \varepsilon) - \entityref{op-function-application}{f}(x_0))$. Тогда для любого $y \in (y_0 - \delta, y_0 + \delta)$ выполняется неравенство $\entityref{op-inverse-function}{g}(y_0) - \varepsilon < \entityref{op-inverse-function}{g}(y) < \entityref{op-inverse-function}{g}(y_0) + \varepsilon$, что доказывает непрерывность $\entityref{op-inverse-function}{g}$ в точке $y_0$.
\end{proof}

\begin{proof}[EN]

It follows that for any $y \in (y_0 - \delta, y_0 + \delta)$, the inequality $\entityref{op-inverse-function}{g}(y_0) - \varepsilon < \entityref{op-inverse-function}{g}(y) < \entityref{op-inverse-function}{g}(y_0) + \varepsilon$ holds, which confirms the continuity of $\entityref{op-inverse-function}{g}$ at $y_0$.
\end{proof}
