\begin{theorem}[Rolle's Theorem]
\mForall{\entityref{obj-function}{f} \colon \entityref{obj-closed-interval}{[a, b]} \mTo \entityref{obj-real-numbers}{\mathbb{R}}}
\left( \entityref{prop-continuous}{f} \text{ on } \entityref{obj-closed-interval}{[a, b]} \right) \mAnd
\left( \entityref{prop-differentiable}{f} \text{ on } \entityref{obj-open-interval}{(a, b)} \right) \mAnd
\left( f(a) = f(b) \right)
\mImplies \mExists{c \in \entityref{obj-open-interval}{(a, b)}} \left( \entityref{obj-function}{f}'(c) = 0 \right)
\end{theorem}

\begin{proof}[RU]
Пусть \entityref{obj-function}{f} непрерывна на \entityref{obj-closed-interval}{[a, b]}. Согласно \entityref{thm-weierstrass-extreme}{теореме Вейерштрасса}, \entityref{obj-function}{f} достигает своих \entityref{obj-extremum}{экстремумов} на \entityref{obj-closed-interval}{[a, b]}. Если \entityref{obj-function}{f} постоянна, то \entityref{obj-derivative}{f'} тождественно равна 0. Если \entityref{obj-function}{f} не постоянна, то хотя бы один из \entityref{obj-extremum}{экстремумов} достигается в некоторой точке \entityref{obj-point}{c} \in \entityref{obj-open-interval}{(a, b)}. В этой точке \entityref{obj-derivative}{f'(c) = 0} согласно \entityref{prop-fermat-lemma}{лемме Ферма}.
\end{proof}

\begin{proof}[EN]
Let \entityref{obj-function}{f} be continuous on \entityref{obj-closed-interval}{[a, b]}. By the \entityref{thm-weierstrass-extreme}{Weierstrass Extreme Value Theorem}, \entityref{obj-function}{f} attains its \entityref{obj-extremum}{extrema} on \entityref{obj-closed-interval}{[a, b]}. If \entityref{obj-function}{f} is constant, then \entityref{obj-derivative}{f'} is identically 0. If \entityref{obj-function}{f} is not constant, at least one \entityref{obj-extremum}{extremum} must be attained at some \entityref{obj-point}{c} \in \entityref{obj-open-interval}{(a, b)}. At this point, \entityref{obj-derivative}{f'(c) = 0} by \entityref{prop-fermat-lemma}{Fermat's Lemma}.
\end{proof}
