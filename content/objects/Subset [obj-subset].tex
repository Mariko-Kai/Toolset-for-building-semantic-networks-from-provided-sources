\section{Подмножество (Subset)}
\label{entity:obj-subset}

\textbf{Описание:}
\entityref{obj-set}{Множество} $A$ называется под\entityref{obj-set}{множеством} \entityref{obj-set}{множества} $B$, если каждый элемент $A$ является элементом $B$.

\hypertarget{obj-subset}{}
\hypertarget{obj-subset}{}
\begin{object}[obj-subset]
\[
A \subset B \mDefIff \mForall{x} (x \in A \mImplies x \in B)
\]
\end{object}
