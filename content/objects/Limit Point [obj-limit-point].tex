\section{Предельная точка множества (Limit Point of a Set)}
\label{entity:obj-limit-point}

\textbf{Описание:}
Точка $p$ называется предельной точкой \entityref{obj-set}{множества} $D$, если в любой окрестности точки $p$ содержится точка из $D$, отличная от $p$.

\entityref{op-hypertarget}{\hypertarget}{obj-limit-point}{}
\hypertarget{obj-limit-point}{}
\begin{object}[obj-limit-point]
\[
\mForall{D \subseteq \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{p \in \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsLimitPoint}(p, D) \mDefinedAs
\mForall{\varepsilon > 0} \mExists{x \in D}
\left( 0 < \entityref{op-abs-abstract}{\mathrm{abs}}(x - p) < \varepsilon \right)
\]
\end{object}
