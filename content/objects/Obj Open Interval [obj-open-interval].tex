\begin{object}[Open Interval]
\[
\mForall{a, b \in \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{S \subseteq \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsOpenInterval}(S, a, b) \mDefinedAs \left( a < b 
\entityref{term-conjunction}{\land} S = \{x \in \entityref{obj-real-numbers}{\mathbb{R}} \entityref{term-mid}{\mid} a < x < b\} \right)
\]
