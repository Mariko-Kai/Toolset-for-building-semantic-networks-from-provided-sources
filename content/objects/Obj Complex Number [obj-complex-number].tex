\begin{object}[Complex Number]
\mForall{z} \mathrm{IsComplexNumber}(z) \mDefinedAs \mExists{a \in \entityref{obj-real-numbers}{\mathbb{R}}} \mExists{b \in \entityref{obj-real-numbers}{\mathbb{R}}} \left( z = \entityref{op-langle}{\langle} a, b \entityref{op-rangle}{\rangle} \right) \mAnd \left( \mathrm{Re}(z) = a \right) \mAnd \left( \mathrm{Im}(z) = b \right)
\end{object}
