\section{Декартово произведение (Cartesian Product)}
\label{entity:obj-cartesian-product}

\textbf{Описание:}
Декартовым произведением \entityref{obj-set}{множеств} $A$ и $B$ называется \entityref{obj-set}{множество} всех \entityref{obj-ordered-pair}{упорядоченных пар} $(a, b)$, где $a \in A$ и $b \in B$.

\entityref{op-hypertarget}{\hypertarget}{obj-cartesian-product}{}
\hypertarget{obj-cartesian-product}{}
\begin{object}[obj-cartesian-product]
\[
\mForall{A \in \entityref{obj-set}{\text{Set}}, B \in \entityref{obj-set}{\text{Set}}}
\quad
A \times B \mDefIff \left\{ (a, b) \entityref{op-mid}{\mid} a \in A \entityref{term-conjunction}{\land} b \in B \right\}
\]
\end{object}
