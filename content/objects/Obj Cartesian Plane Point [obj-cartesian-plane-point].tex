\begin{object}[CartesianPlanePoint]
\mForall{a \colon \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{b \colon \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsPoint}(a, b, p) \mDefinedAs
\left( p = (a, b) \right) \mAnd
\left( p \in \entityref{obj-cartesian-product}{\mathbb{R} \times \mathbb{R}} \right) \mAnd
\left( \exists O, X, Y \in \entityref{obj-cartesian-product}{\mathbb{R} \times \mathbb{R}} \right)
\left( O = (0, 0) \right) \mAnd
\left( X = (a, 0) \right) \mAnd
\left( Y = (0, b) \right) \mAnd
\left( p \text{ is the vertex of the rectangle with vertices } \{O, X, Y, p\} \right)
\end{object}
