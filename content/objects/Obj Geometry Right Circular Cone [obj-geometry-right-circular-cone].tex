\begin{object}[Right Circular Cone]
\mForall{P \in \entityref{obj-euclidean-space-3d}{V^3}}
\mForall{A \in \entityref{obj-euclidean-space-3d}{V^3} \setminus \{0\}}
\mForall{\theta \in \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsRightCircularCone}(S, P, A, \theta) \mDefinedAs
\left( S = \{ X \in \entityref{obj-euclidean-space-3d}{V^3} \mid \mExists{L \in \entityref{obj-set-of-lines}{Lines(P)}} 
(X \in L) \mAnd (\mathrm{Angle}(L, A) = \theta) \} \right)
\end{object}
