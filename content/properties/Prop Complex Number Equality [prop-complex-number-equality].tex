\begin{property}[Equality of Complex Numbers]
\mForall{z_1 \in \entityref{obj-complex-numbers}{\mathbb{C}}}
\mForall{z_2 \in \entityref{obj-complex-numbers}{\mathbb{C}}}
\mathrm{IsEqual}(z_1, z_2) \mDefinedAs 
\left( z_1 = (a, b) \right) \mAnd \left( z_2 = (c, d) \right) \mAnd (a = c) \mAnd (b = d)
\end{property}
