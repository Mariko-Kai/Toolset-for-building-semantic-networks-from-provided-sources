\begin{property}
\mForall{f \colon \entityref{obj-closed-interval}{[a, b]} \mTo \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsBounded}(f) \mDefinedAs \mExists{M \in \entityref{obj-real-numbers}{\mathbb{R}}} \left( (M > 0) \mAnd \mForall{x \in \entityref{obj-closed-interval}{[a, b]}} \left( \entityref{op-abs-abstract}{\mathrm{abs}}(f(x)) < M \right) \right)
\end{property}
