\begin{property}[Uniform Convergence]
\[
\mForall{S \subseteq \entityref{obj-complex-numbers}{\mathbb{C}}}
\mForall{f_n \colon S \mTo \entityref{obj-complex-numbers}{\mathbb{C}}}
\mForall{\entityref{obj-function}{f} \colon S \mTo \entityref{obj-complex-numbers}{\mathbb{C}}}
\quad (\text{uniform-convergence}(f_n, \entityref{obj-function}{f}, S)) \mIff
\mForall{\varepsilon \in \entityref{obj-real-numbers}{\mathbb{R}}} \quad \varepsilon > 0 \mImplies
\mExists{N \in \entityref{obj-natural-numbers}{\mathbb{N}}} \quad \mForall{n \in \entityref{obj-natural-numbers}{\mathbb{N}}} \quad n > N \mImplies
\mForall{x \in S} \quad \entityref{op-abs-abstract}{\mathrm{abs}}(f_n(x) - f(x)) < \varepsilon
\]
\end{property}
