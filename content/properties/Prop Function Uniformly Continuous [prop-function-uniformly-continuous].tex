\begin{property}[UniformlyContinuous]
    \begin{aligned}
\mForall{E \subseteq \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{f \colon E \mTo \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsUniformlyContinuous}(f, E) \mDefinedAs
\mForall{\varepsilon \in \entityref{obj-real-numbers}{\mathbb{R}}}
\left( \varepsilon > 0 \mImplies \mExists{\delta \in \entityref{obj-real-numbers}{\mathbb{R}}} \left( \delta > 0 \mAnd \mForall{x_1 \in E} \mForall{x_2 \in E} \left( \entityref{op-abs-abstract}{\mathrm{abs}}(x_1 - x_2) < \delta \mImplies \entityref{op-abs-abstract}{\mathrm{abs}}(f(x_1) - f(x_2)) < \varepsilon \right) \right) \right)
    \end{aligned}
\end{property}
