\section{Предел функции (Limit of a Function)}
\label{entity:prop-limit-function}

\textbf{Описание:}
Число $A$ называется пределом \entityref{obj-function}{функции} $\entityref{obj-function}{f}$ при $x$, стремящемся к $a$ (по \entityref{obj-set}{множеству} $E$), если для любого $\varepsilon > 0$ существует такое $\delta > 0$, что для любой точки $x \in E$, удовлетворяющей условию $0 < |x - a| < \delta$, выполнено неравенство $|f(x) - A| < \varepsilon$.

\entityref{op-hypertarget}{\hypertarget}{prop-limit-function}{}
\hypertarget{prop-limit-function}{}
\begin{property}[Limit of a Function]
\[
\mForall{E \subseteq \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{a \in \entityref{obj-limit-point}{E'}}
\mForall{\entityref{obj-function}{f} \colon E \mTo \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{A \in \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsLimit}(\entityref{obj-function}{f}, a, A) \mDefinedAs
\mForall{\varepsilon > 0} \mExists{\delta > 0} \mForall{x \in E}
\left( 0 < \entityref{op-abs-abstract}{\mathrm{abs}}(x - a) < \delta \mImplies \entityref{op-abs-abstract}{\mathrm{abs}}(f(x) - A) < \varepsilon \right)
\]
\end{property}
