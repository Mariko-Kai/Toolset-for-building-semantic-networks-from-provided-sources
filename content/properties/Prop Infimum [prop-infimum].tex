\begin{property}
\mForall{S \subseteq \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{L \in \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsInfimum}(L, S) \mDefinedAs 
\left( \mForall{s \in S} (L \leq s) \right) \mAnd 
\mNot \left( \mExists{L' \in \entityref{obj-real-numbers}{\mathbb{R}}} (L' > L \mAnd \mForall{s \in S} (L' \leq s)) \right)
\end{property}
