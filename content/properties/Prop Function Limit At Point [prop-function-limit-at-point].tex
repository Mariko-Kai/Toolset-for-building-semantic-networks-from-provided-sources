\begin{property}[Limit of a Function at a Point]
\mForall{f \colon D \mTo \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{p \colon \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{A \colon \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{HasLimit}(f, p, A) \mDefinedAs
\left( \mForall{\varepsilon \in \entityref{obj-real-numbers}{\mathbb{R}}} \quad \varepsilon > 0 \mImplies
\mExists{\delta \in \entityref{obj-real-numbers}{\mathbb{R}}} \quad \delta > 0 \mAnd
\left( \mForall{x \in D} \quad 0 < \entityref{op-abs-abstract}{\mathrm{abs}}(x - p) < \delta \mImplies \entityref{op-abs-abstract}{\mathrm{abs}}(f(x) - A) < \varepsilon \right) \right)
\end{property}
