\begin{property}[Neighborhood]
\mForall{p \in \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{\varepsilon \in \entityref{obj-real-numbers}{\mathbb{R}}, \varepsilon > 0}
\mForall{N \subseteq \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsNeighborhood}(N, p) \mDefinedAs \mExists{\varepsilon > 0} \left( N = \left( p - \varepsilon, p + \varepsilon \right) \right)
\end{property}
