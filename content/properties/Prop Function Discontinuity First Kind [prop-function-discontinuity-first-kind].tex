\begin{property}[Discontinuity of the First Kind]
\mForall{E \subseteq \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{f \colon E \mTo \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{a \in E}
\mathrm{IsDiscontinuityFirstKind}(f, a) \mDefinedAs
\left( \mExists{L_- \in \entityref{obj-real-numbers}{\mathbb{R}}} \left( L_- = \entityref{op-limit}{\lim}_{E \ni x \to a-0} f(x) \right) \right)
\mAnd
\left( \mExists{L_+ \in \entityref{obj-real-numbers}{\mathbb{R}}} \left( L_+ = \entityref{op-limit}{\lim}_{E \ni x \to a+0} f(x) \right) \right)
\mAnd
\left( L_- \neq f(a) \mOr L_+ \neq f(a) \right)
\end{property}
