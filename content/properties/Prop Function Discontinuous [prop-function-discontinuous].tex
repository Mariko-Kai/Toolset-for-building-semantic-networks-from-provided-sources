\begin{property}[Discontinuity]
\mForall{D \colon \mathrm{Type}}
\mForall{f \colon D \mTo \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{p \colon D}
\mathrm{IsDiscontinuous}(f, p) \mDefinedAs \mNot \left( \entityref{prop-function-continuous-at-point}{f, p} \right)
\mAnd \left( \mNot \left( \exists{y \colon \entityref{obj-real-numbers}{\mathbb{R}}} \left( \entityref{op-limit}{\lim}_{x \to p} f(x) = y \right) \right) \mOr \left( \entityref{op-limit}{\lim}_{x \to p} f(x) \neq f(p) \right) \right)
\end{property}
