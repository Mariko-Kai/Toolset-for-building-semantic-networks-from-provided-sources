\begin{property}
\mForall{f \colon D \to \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{a \in D}
\mathrm{IsRemovableDiscontinuity}(f, a) \mDefinedAs
\left( \mExists{L \in \entityref{obj-real-numbers}{\mathbb{R}}} \left( L = \entityref{op-limit}{\lim}_{x \to a} f(x) \right) \right) \mAnd \left( \entityref{op-limit}{\lim}_{x \to a} f(x) \neq f(a) \right)
\end{property}
