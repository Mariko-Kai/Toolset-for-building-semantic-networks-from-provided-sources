\begin{property}[Continuous]
\[
\mForall{\entityref{obj-function}{f} \colon D \mTo \entityref{obj-real-numbers}{\mReal}}
\mForall{x_0 \in D}
\mathrm{IsContinuous}(\entityref{obj-function}{f}, x_0) \mDefinedAs
\mForall{\varepsilon > 0}
\mExists{\delta > 0}
\mForall{x \in D}
\left( \entityref{op-abs-abstract}{\mathrm{abs}}(x - x_0) < \delta \right)
\mImplies
\left( \entityref{op-abs-abstract}{\mathrm{abs}}(f(x) - f(x_0)) < \varepsilon \right)
\]
\end{property}
