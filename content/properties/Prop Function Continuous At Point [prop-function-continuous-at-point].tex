\begin{property}[Continuous at a point]
\mForall{f \colon \entityref{obj-real-numbers}{\mathbb{R}} \mTo \entityref{obj-real-numbers}{\mathbb{R}}}
\mForall{p \colon \entityref{obj-real-numbers}{\mathbb{R}}}
\mathrm{IsContinuousAt}(f, p) \mDefinedAs 
\left( \entityref{op-limit}{\lim}_{x \to p} f(x) = f(p) \right)
\mAnd
\left( \mForall{\varepsilon \in \entityref{obj-real-numbers}{\mathbb{R}}} \quad \varepsilon > 0 \mImplies \mExists{\delta \in \entityref{obj-real-numbers}{\mathbb{R}}} \quad \delta > 0 \mAnd \mForall{x \in \entityref{obj-real-numbers}{\mathbb{R}}} \quad \entityref{op-abs-abstract}{\mathrm{abs}}(x - p) < \delta \mImplies \entityref{op-abs-abstract}{\mathrm{abs}}(f(x) - f(p)) < \varepsilon \right)
\end{property}
