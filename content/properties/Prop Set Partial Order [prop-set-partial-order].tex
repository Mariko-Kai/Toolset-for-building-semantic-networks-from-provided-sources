\begin{property}[Partial Order]
\mForall{X \colon \mathrm{Type}}
\mForall{R \colon X \mTo X \mTo \mathrm{Prop}}
\mathrm{IsPartialOrder}(X, R) \mDefinedAs
\left(
\begin{aligned}
& \left( \mForall{x \colon X} \entityref{prop-relation-reflexive}{R}(x, x) \right) \mAnd \\
& \left( \mForall{x \colon X} \mForall{y \colon X} \mForall{z \colon X} \left( R(x, y) \mAnd R(y, z) \mImplies R(x, z) \right) \right) \mAnd \\
& \left( \mForall{x \colon X} \mForall{y \colon X} \left( R(x, y) \mAnd R(y, x) \mImplies x = y \right) \right)
\end{aligned}
\right)
\end{property}
