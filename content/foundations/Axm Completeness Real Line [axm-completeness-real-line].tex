\begin{axiom}[Completeness of the Real Line]
\[
\mForall{L} \mathrm{IsComplete}(L) \mDefinedAs \mNot 
\mExists{A, B \subset L} \left( (A \neq 
\entityref{obj-empty-set}{\\varnothing}) \
entityref{term-conjunction}{\land} (B \neq 
\entityref{obj-empty-set}{\\varnothing}) 
\entityref{term-conjunction}{\land} (A 
\entityref{op-union}{\cup} B = L) 
\entityref{term-conjunction}{\land} (A 
\entityref{op-intersection}{\cap} B = \entityref{obj-empty-set}{\\varnothing}) \entityref{term-conjunction}{\land} \mForall{a \in A} \mForall{b \in B} (a < b) \entityref{term-conjunction}{\land} \mNot \mExists{p \in L} ((p \in A) \entityref{term-disjunction}{\lor} (p \in B)) \right)
\]
\end{axiom}
