\begin{axiom}[axm-zfc-replacement] % Аксиома подстановки (Axiom of Replacement)



\[
\left( \mForall{x} \mExists{! y} \, \phi(x, y) \right)
\mImplies
\mForall{X} \mExists{Y} \mForall{y} \left( y \in Y \mIff \mExists{x} \left( x \in X \entityref{term-conjunction}{\land} \phi(x, y) \right) \right)
\]
\end{axiom}
