\begin{axiom}[axm-zfc-regularity] % Аксиома регулярности (Axiom of Regularity)



\[
\mForall{S}
\left( (\mExists{y} y \in S) \mImplies \mExists{x} \left( x \in S \entityref{term-conjunction}{\land} \lnot \mExists{z} (z \in S \entityref{term-conjunction}{\land} z \in x) \right) \right)
\]
\end{axiom}
