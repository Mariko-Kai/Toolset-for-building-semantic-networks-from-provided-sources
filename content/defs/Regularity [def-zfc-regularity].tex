\begin{axiom}[axm-zfc-regularity] % Аксиома регулярности (Axiom of Regularity)



\[
\TerForall S
\left( (\TermExists y y \in S) \TermImplication \TermExists x \left( x \in S \TermConjunction{\land} \lnot \TermExists z (z \in S \TermConjunction{\land} z \in x) \right) \right)
\]
\end{axiom}
