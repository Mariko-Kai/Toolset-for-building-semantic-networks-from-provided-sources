\begin{definition}[def-complex-number]
\[
\TerForall z \mathrm{IsComplexNumber}(z) \TermDefinedAs \TermExists a \in \RealNumbers \TermExists b \in \RealNumbers \left( z = \langle a, b \rangle \right) \TermConjunction \left( \mathrm{Re}(z) = a \right) \TermConjunction \left( \mathrm{Im}(z) = b \right)
\]
\end{definition}
