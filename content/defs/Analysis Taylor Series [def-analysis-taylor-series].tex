\begin{definition}[Taylor Series]
\[
\TerForall f \colon \RealNumbers \TermTo \RealNumbers \TerForall a \colon \RealNumbers \TerForall x \colon \RealNumbers
\mathrm{IsTaylorSeries}(f, a, x) \coloneqq \sum_{k=0}^{\infty} \frac{f^{(k)}(a)}{k!} \cdot (x - a)^k
\]
\end{definition}
