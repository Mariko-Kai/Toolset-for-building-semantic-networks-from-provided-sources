\begin{definition}[def-equivalence] % Эквивалентность (Equivalence)

\textbf{Символ:} $\iff$ (также $\TermEquivalence{\leftrightarrow}$)

\textbf{Читается как:} «тогда и только тогда, когда», «эквивалентно».

\textbf{Формальная семантика:}
Пусть $P$ и $Q$ — утверждения. Утверждение $P \iff Q$ истинно тогда и только тогда, когда утверждения $P$ и $Q$ имеют одинаковые истинностные значения (оба истинны или оба ложны).

\textbf{Пример:} $(2 < 3) \iff (3 > 2)$
\end{definition}
