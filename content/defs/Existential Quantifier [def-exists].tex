\begin{definition}[def-exists] % Квантор существования (Existential Quantifier)

\textbf{Символ:} $\exists$

\textbf{Читается как:} «существует», «найдется».

\textbf{Формальная семантика:}
Пусть $P(x)$ — предикат. Утверждение $\exists x\, P(x)$ истинно тогда и только тогда,
когда в области дискурса (универсума) существует хотя бы один элемент $x$, для которого $P(x)$ истинно.

\textbf{Пример:} $\exists x \in \RealNumbers\; (x^2 = 2)$
\end{definition}
