\begin{axiom}[axm-zfc-infinity] % Аксиома бесконечности (Axiom of Infinity)



\[
\TermExists I
\left( \ZfcPairing{\varnothing} \in I \;\TermConjunction{\land}\; \TerForall x \left( x \in I \TermImplication \TermExists y (y \in I \TermConjunction{\land} \TerForall z (z \in y \TermEquivalence z \in x \TermDisjunction{\lor} z = x)) \right) \right)
\]
\end{axiom}
