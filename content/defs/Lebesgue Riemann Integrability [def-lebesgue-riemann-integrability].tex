\begin{property}[Lebesgue Criterion for Riemann Integrability]
\[
\mathrm{IsRiemannIntegrable}(f, [a, b]) \TermDefinedAs \left( \FunctionBounded{f} \TermConjunction \mu(\{x \in [a, b] \mid \FunctionDiscontinuous{f}(x)\}) = 0 \right)
\]
\end{property}
