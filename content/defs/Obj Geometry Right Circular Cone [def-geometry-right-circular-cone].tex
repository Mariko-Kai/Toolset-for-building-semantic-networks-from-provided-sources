\hypertarget{def-geometry-right-circular-cone}{}
\begin{object}[Right Circular Cone]
\[
\TerForall P \in V^3
\TerForall A \in V^3 \setminus \{0\}
\TerForall \theta \in \RealNumbers
\mathrm{IsRightCircularCone}(S, P, A, \theta) \TermDefinedAs
\left( S = \{ X \in V^3 \mid \TermExists L \in \Set{Lines(P)} 
(X \in L) \And (\mathrm{Angle}(L, A) = \theta) \} \right)
\]
\end{object}
