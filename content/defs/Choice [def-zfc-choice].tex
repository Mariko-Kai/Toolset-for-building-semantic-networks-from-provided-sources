\begin{axiom}[axm-zfc-choice] % Аксиома выбора (Axiom of Choice)



\[
\TerForall \mathcal{F}
\left(
  \TerForall A, B \left( A \in \mathcal{F} \TermConjunction{\land} B \in \mathcal{F} \TermConjunction{\land} A \neq B \TermImplication \lnot \TermExists x (x \in A \TermConjunction{\land} x \in B) \right)
  \TermConjunction{\land} \lnot(\ZfcPairing{\varnothing} \in \mathcal{F})
\right)
\TermImplication
\TermExists C \TerForall A \left( A \in \mathcal{F} \TermImplication \TermExists ! x (x \in A \TermConjunction{\land} x \in C) \right)
\]
\end{axiom}
