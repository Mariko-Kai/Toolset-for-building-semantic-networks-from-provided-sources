\begin{definition}[\mathrm{IsInfinitesimal}]
\terForall f \colon \Function{\RealNumbers}{\RealNumbers} \mathrm{IsInfinitesimal}(f) \coloneqq \left( \forall \epsilon \colon \RealNumbers \left( \epsilon > 0 \Rightarrow \AbsAbstract{f} < \epsilon \right) \right)
\end{definition}
