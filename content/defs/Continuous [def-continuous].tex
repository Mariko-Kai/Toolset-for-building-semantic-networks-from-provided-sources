\begin{definition}[def-continuous]
\[
\TerForall \Function{f} \colon D \TermTo \RealNumbers
\TerForall x_0 \in D
\mathrm{IsContinuous}(\Function{f}, x_0) \TermDefinedAs
\TerForall \varepsilon > 0
\TermExists \delta > 0
\TerForall x \in D
\left( \AbsAbstract{ x - x_0 } < \delta \right)
\TermImplication
\left( \AbsAbstract{ f(x) - f(x_0) } < \varepsilon \right)
\]
\end{definition}
