\begin{property}[Equality of Complex Numbers]
\[
\TerForall z_1 \in \ComplexNumber
\TerForall z_2 \in \ComplexNumber
\mathrm{IsEqual}(z_1, z_2) \TermDefinedAs 
\left( z_1 = (a, b) \right) \TermConjunction \left( z_2 = (c, d) \right) \TermConjunction (a = c) \TermConjunction (b = d)
\]
\end{property}
