\section{Аксиома полноты (Completeness Axiom)}

\hypertarget{axm-completeness}{}
\begin{axiom}[axm-completeness]
\label{entity:axm-completeness}
% 1. ОБЪЯВЛЕНИЕ ТИПОВ
$\Set{X, Y} \subset \RealNumbers$

% 2. ТЕЛО АКСИОМЫ
\[
(\TerForall x \in \Set{X} \TermConjunction \TerForall y \in \Set{Y} \colon x \SetPartialOrder{\le} y) \TermImplication 
(\TermExists c \in \RealNumbers \TerForall x \in \Set{X}, \TerForall y \in \Set{Y} \colon x \SetPartialOrder{\le} c \SetPartialOrder{\le} y)
\]
\end{axiom}
