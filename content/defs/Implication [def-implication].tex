\begin{definition}[def-implication] % Импликация (Implication)

\textbf{Символ:} $\implies$ (также $\to$)

\textbf{Читается как:} «если..., то...», «влечет».

\textbf{Формальная семантика:}
Пусть $P$ и $Q$ — утверждения. Утверждение $P \implies Q$ ложно тогда и только тогда, когда $P$ истинно, а $Q$ ложно. В остальных случаях импликация истинна.

\textbf{Пример:} $(2 < 1) \implies (3 = 4)$
\end{definition}
