\section{Предельная точка множества (Limit Point of a Set)}
\label{entity:obj-limit-point}

\textbf{Описание:}
Точка $p$ называется предельной точкой \Set{множества} $D$, если в любой окрестности точки $p$ содержится точка из $D$, отличная от $p$.

\hypertarget{def-limit-point}{}
\begin{definition}[def-limit-point]
\[
\TerForall D \subseteq \RealNumbers
\TerForall p \in \RealNumbers
\mathrm{IsLimitPoint}(p, D) \TermDefinedAs
\TerForall \varepsilon > 0 \TermExists x \in D
\left( 0 < \AbsAbstract{ x - p } < \varepsilon \right)
\]
\end{definition}
