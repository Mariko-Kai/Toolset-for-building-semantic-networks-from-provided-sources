\begin{definition}[def-set-equality]
\[
\forall A \colon \Set \quad \forall B \colon \Set \quad \mathrm{IsEqualSet}(A, B) \coloneqq \forall x \colon \Universe \left( (x \in A) \iff (x \in B) \right)
\]
\end{definition}
