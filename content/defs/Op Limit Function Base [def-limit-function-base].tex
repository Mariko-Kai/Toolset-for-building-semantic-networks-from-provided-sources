\section{Предел функции по базе (Limit of a Function over a Base)}
\label{entity:op-limit-function-base}

\textbf{Описание:}
Число $A$ называется пределом \Function{функции} $\Function{f}$ по базе $\mathcal{B}$, если для любого $\varepsilon > 0$ существует такой элемент $B \in \mathcal{B}$ базы $\mathcal{B}$, что для всех $x \in B$ выполняется неравенство $|f(x) - A| < \varepsilon$.

\hypertarget{def-limit-function-base}{}
\begin{operation}
\[
\TerForall X \in \mathrm{Set}
\TerForall \mathcal{B} \subseteq \mathcal{P}(X)
\TerForall \Function{f} \colon X \TermTo \RealNumbers
\TerForall A \in \RealNumbers
\mathrm{IsLimitBase}(\Function{f}, \mathcal{B}, A) \TermDefinedAs
\TerForall \varepsilon > 0 \TermExists B \in \mathcal{B} \TerForall x \in B
\AbsAbstract{\left| f(x) - A \right|} < \varepsilon
\]
\end{operation}
