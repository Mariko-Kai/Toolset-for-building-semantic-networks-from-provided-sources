\section{Последовательность (Sequence)}
\label{entity:obj-sequence}

\textbf{Описание:}
Последовательность элементов \Set{множества} $X$ — это \Function{функция}, определенная на \Set{множестве} натуральных чисел $\NaturalNumbers$ со значениями в $X$.

\hypertarget{def-sequence}{}
\begin{definition}[def-sequence]
\Set{X} \text{ --- произвольное \Set{множество}}

\[
(x_n)_{n \in \NaturalNumbers} \TermDefIff \Function{f} \colon \NaturalNumbers \to X
\]
\end{definition}
