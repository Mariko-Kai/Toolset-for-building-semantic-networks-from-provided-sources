\begin{definition}[def-zfc-pairing] % Аксиома пары (Axiom of Pairing)



\[
\TerForall A, B
\TermExists C \TerForall x \left( x \in C \TermEquivalence (x = A \TermDisjunction{\lor} x = B) \right)
\]
