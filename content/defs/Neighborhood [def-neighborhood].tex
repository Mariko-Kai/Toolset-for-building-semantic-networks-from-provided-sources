\begin{definition}[def-neighborhood]
\[
\TerForall p \in \RealNumbers
\TerForall \varepsilon \in \RealNumbers, \varepsilon > 0
\TerForall N \subseteq \RealNumbers
\mathrm{IsNeighborhood}(N, p) \TermDefinedAs \TermExists \varepsilon > 0 \left( N = \left( p - \varepsilon, p + \varepsilon \right) \right)
\]
\end{definition}
