\begin{operation}
\[
\mathrm{IsSupremum}(s, X) \TermDefinedAs \left( \TerForall{\forall} x \in X (x \leq s) \TermConjunction{\land} \TerForall{\forall} s' < s (\TermExists{\exists} x' \in X (s' < x')) \right)
\]
\end{operation}
