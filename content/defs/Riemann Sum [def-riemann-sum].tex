\hypertarget{def-riemann-sum}{}
\begin{definition}
\[
\TerForall f \colon \RealNumbers \TermTo \RealNumbers, \TerForall a \colon \RealNumbers, \TerForall b \colon \RealNumbers, \TerForall n \colon \NaturalNumbers, \TerForall P \colon \ClosedInterval{a, b}^{n+1}, \TerForall x \colon \NaturalNumbers \TermTo \RealNumbers
\]
\[
\mathrm{IsRiemannSum}(f, P, x, n, S) \coloneqq \left( S = \sum_{i=1}^{n} f(x_i) \cdot (t_i - t_{i-1}) \right)
\]
\end{definition}
