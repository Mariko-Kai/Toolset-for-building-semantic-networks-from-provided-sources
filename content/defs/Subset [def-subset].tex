\section{Подмножество (Subset)}
\label{entity:obj-subset}

\textbf{Описание:}
\Set{Множество} $A$ называется под\Set{множеством} \Set{множества} $B$, если каждый элемент $A$ является элементом $B$.

\hypertarget{def-subset}{}
\begin{definition}[def-subset]
Пусть
\[
A \subset B \TermDefIff \TerForall x (x \in A \TermImplication x \in B)
\]
\end{definition}
