\section{Декартово произведение (Cartesian Product)}
\label{entity:obj-cartesian-product}

\textbf{Описание:}
Декартовым произведением \Set{множеств} $A$ и $B$ называется \Set{множество} всех \OrderedPair{упорядоченных пар} $(a, b)$, где $a \in A$ и $b \in B$.

\hypertarget{def-cartesian-product}{}
\begin{definition}[def-cartesian-product]
\[
\TerForall A \in \Set{\text{Set}}, B \in \Set{\text{Set}}
\quad
A \times B \TermDefIff \left\{ (a, b) \TermMid{\mid} a \in A \TermConjunction{\land} b \in B \right\}
\]
\end{definition}
