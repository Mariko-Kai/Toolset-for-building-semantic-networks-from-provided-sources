\begin{definition}[Equivalence Relation]
\TerForall A \colon \Set
\TerForall R \colon \SetSubset{\CartesianProduct{A}{A}}
\mathrm{IsEquivalenceRelation}(R) \coloneqq \left( \forall x \colon A, (x, x) \in R \right) \TermConjunction \left( \forall x \colon A, \forall y \colon A, ((x, y) \in R \Rightarrow (y, x) \in R) \right) \TermConjunction \left( \forall x \colon A, \forall y \colon A, \forall z \colon A, (((x, y) \in R \land (y, z) \in R) \Rightarrow (x, z) \in R) \right)
\end{definition}
