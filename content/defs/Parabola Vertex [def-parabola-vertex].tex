\begin{definition}[def-parabola-vertex]
\[
\TerForall F \colon \RealNumbers^2
\TerForall L \colon \mathcal{L}
\TerForall V \colon \RealNumbers^2
\mathrm{IsVertex}(V, F, L) \TermDefinedAs 
\left( V = \frac{1}{2} \left( F + P_L(F) \right) \right)
\]
\end{definition}
