\section{Частичный порядок (Partial Order)}

\hypertarget{def-relation-partial-order}{}
\begin{object}[Partial Order]
\[
\label{entity:obj-relation-partial-order}
% 1. ОБЪЯВЛЕНИЕ ТИПОВ
\TerForall X \colon \Set{\mathrm{Set}}
\TerForall R \subset \CartesianProduct{X \times X}

% 2. ТЕЛО ОПРЕДЕЛЕНИЯ
\mathrm{IsPartialOrder}(R, X) \TermDefinedAs \left(
    \RelationPartialOrder{\mathrm{IsReflexive}}(R, X) \And
    \RelationPartialOrder{\mathrm{IsAntisymmetric}}(R, X) \And
    \RelationPartialOrder{\mathrm{IsTransitive}}(R, X)
\right)
\]
