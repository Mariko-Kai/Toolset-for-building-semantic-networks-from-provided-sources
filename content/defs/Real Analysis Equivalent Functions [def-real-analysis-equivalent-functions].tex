\begin{definition}[Equivalent Functions]
\mathrm{IsEquivalentFunction}(f, g) \coloneqq \mathrm{TermExists} C \colon \RealNumbers \cdot \mathrm{TermForall} x \colon \RealNumbers \cdot \left( f(x) = g(x) + C \right)
\end{definition}
