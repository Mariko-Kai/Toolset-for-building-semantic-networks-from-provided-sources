% entity-id: def-real-analysis-limitAtInfinity
% entity-type: def
% macro: IsLimitAtPlusInfinity
% args: 1
% notation: \mathrm{IsLimitAtPlusInfinity}(f)
\begin{definition}[Limit at Positive Infinity]
\mathrm{IsLimitAtPlusInfinity}(f) \coloneqq \forall \epsilon \colon \RealNumbers \ \exists M \colon \RealNumbers \ \forall x \colon \RealNumbers \ ( \mathrm{RealAbsoluteValue{x}} > M \ \Rightarrow \ f(x) > \epsilon )$
\end{definition}
