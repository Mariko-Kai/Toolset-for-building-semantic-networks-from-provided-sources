\section{Предел функции (Limit of a Function)}
\label{entity:prop-limit-function}

\textbf{Описание:}
Число $A$ называется пределом \Function{функции} $\Function{f}$ при $x$, стремящемся к $a$ (по \Set{множеству} $E$), если для любого $\varepsilon > 0$ существует такое $\delta > 0$, что для любой точки $x \in E$, удовлетворяющей условию $0 < |x - a| < \delta$, выполнено неравенство $|f(x) - A| < \varepsilon$.

\hypertarget{def-limit-function}{}
\begin{property}[Limit of a Function]
\[
\TerForall E \subseteq \RealNumbers
\TerForall a \in \LimitPoint{E'}
\TerForall \Function{f} \colon E \TermTo \RealNumbers
\TerForall A \in \RealNumbers
\mathrm{IsLimit}(\Function{f}, a, A) \TermDefinedAs
\TerForall \varepsilon > 0 \TermExists \delta > 0 \TerForall x \in E
\left( 0 < \AbsAbstract{ x - a } < \delta \TermImplication \AbsAbstract{ f(x) - A } < \varepsilon \right)
\]
\end{property}
