\begin{operation}[Upper Riemann Integral]
\[
\TerForall \Function{f} \colon \ClosedInterval{[a,b]} \TermTo \RealNumbers
\TerForall I \in \RealNumbers
\mathrm{IsUpperIntegral}(\Function{f}, I) \TermDefinedAs \left( I = \Infimum{\inf} \{ \RiemannIntegral{\int}_{a}^{b} \Function{t}(x) \, dx \TermMid{\mid} \Function{t} \geq \Function{f} \} \right)
\]
\end{operation}
