\begin{property}[Partial Order]
\[
\TerForall X \colon \mathrm{Type}
\TerForall R \colon X \TermTo X \TermTo \mathrm{Prop}
\mathrm{IsPartialOrder}(X, R) \TermDefinedAs
\left(
\begin{aligned}
& \left( \TerForall x \colon X \RelationPartialOrder{R}(x, x) \right) \TermConjunction \\
& \left( \TerForall x \colon X \TerForall y \colon X \TerForall z \colon X \left( R(x, y) \TermConjunction R(y, z) \TermImplication R(x, z) \right) \right) \TermConjunction \\
& \left( \TerForall x \colon X \TerForall y \colon X \left( R(x, y) \TermConjunction R(y, x) \TermImplication x = y \right) \right)
\end{aligned}
\right)
\]
\end{property}
