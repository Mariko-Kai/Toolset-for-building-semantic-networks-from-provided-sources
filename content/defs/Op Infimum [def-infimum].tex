\hypertarget{def-infimum}{}
\begin{operation}[Infimum]
\[
\TerForall S \subseteq \RealNumbers
\TerForall L \in \RealNumbers
\mathrm{IsInfimum}(L, S) \TermDefinedAs
\left(
\TerForall s \in S \quad L \leq s
\right)
\And
\left(
\TerForall y \in \RealNumbers
\left(
\left(
\TerForall s \in S \quad y \leq s
\right)
\TermImplication
y \leq L
\right)
\right)
\]
\end{operation}
