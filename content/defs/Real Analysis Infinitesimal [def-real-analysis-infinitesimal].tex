\begin{definition}[Infinitesimal Quantity]
\TerForall x \colon \RealNumbers \mathrm{IsInfinitesimal}(x) \coloneqq \left( x \neq 0 \right) \land \left( \forall r \colon \RealNumbers, \left( r > 0 \Rightarrow \AbsAbstract{x} < r \right) \right)
\end{definition}
