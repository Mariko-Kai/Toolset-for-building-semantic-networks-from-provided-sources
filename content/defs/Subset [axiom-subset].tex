\begin{object}[term-subset] % \SetSubset{Подмножество} (Subset)



\textbf{Символ:} $\subset$ (или $\subseteq$)

\textbf{Формальное определение:}
\Set{Множество} $A$ называется \SetSubset{подмножеством} \Set{множества} $B$, если каждый элемент $A$ является элементом $B$:
\[ A \subset B \TermDefIff \TerForall x (x \in A \TermImplication x \in B) \]

\textbf{Примечание:} В архитектуре Mathesis символ $\subset$ является терминалом и не требует обертки в \texttt{entityref}.
\end{object}
