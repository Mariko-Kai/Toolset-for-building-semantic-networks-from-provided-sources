\begin{axiom}[Completeness of the Real Line]
\[
\TerForall L \mathrm{IsComplete}(L) \TermDefinedAs \TermNegation 
\TermExists A, B \subset L \left( (A \neq 
\Set{\varnothing}) \TermConjunction{\land} (B \neq 
\Set{\varnothing}) 
\TermConjunction{\land} (A 
\cup B = L) 
\TermConjunction{\land} (A 
\cap B = \Set{\varnothing}) \TermConjunction{\land} \TerForall a \in A \TerForall b \in B (a < b) \TermConjunction{\land} \TermNegation \TermExists p \in L ((p \in A) \TermDisjunction{\lor} (p \in B)) \right)
\]
\end{axiom}
