\begin{property}[Limit of a Function at a Point]
\[
\TerForall f \colon D \TermTo \RealNumbers
\TerForall p \colon \RealNumbers
\TerForall A \colon \RealNumbers
\mathrm{HasLimit}(f, p, A) \TermDefinedAs
\left( \TerForall \varepsilon \in \RealNumbers \quad \varepsilon > 0 \TermImplication
\TermExists \delta \in \RealNumbers \quad \delta > 0 \And
\left( \TerForall x \in D \quad 0 < \AbsAbstract{\left| x - p \right|} < \delta \TermImplication \AbsAbstract{\left| f(x) - A \right|} < \varepsilon \right) \right)
\]
\end{property}
