% macro: \DerivativeAtPoint
% args: 1
% notation: #1
\hypertarget{def-derivative-at-point}{}
\begin{operation}
\[
\TerForall \Function{f} \colon \OpenInterval{(a, b)} \TermTo \RealNumbers
\TerForall x \in (a, b)
\TerForall L \in \RealNumbers
\mathrm{IsDerivative}(\Function{f}, x, L) \TermDefinedAs \left( L = \LimitFunctionBase{\lim}_{\Function{h} \to 0} \frac{f(x + \Function{h}) - f(x)}{\Function{h}} \right)
\]
\end{operation}
