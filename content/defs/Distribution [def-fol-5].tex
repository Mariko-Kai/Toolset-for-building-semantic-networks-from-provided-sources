\begin{axiom}[axm-fol-5]
\label{entity:axm-fol-5}
% 1. ОБЪЯВЛЕНИЕ ТИПОВ
$\Predicate{\mathcal{B}}, \Predicate{\mathcal{C}}$ --- формулы логики первого порядка,
переменная $x$ не входит свободно в $\Predicate{\mathcal{B}}$

% 2. ТЕЛО АКСИОМЫ
\[
(\TerForall x (\Predicate{\mathcal{B}} \TermImplication{\to} \Predicate{\mathcal{C}})) \TermImplication{\to} (\Predicate{\mathcal{B}} \TermImplication{\to} (\TerForall x \Predicate{\mathcal{C}}))
\]
\end{axiom}
