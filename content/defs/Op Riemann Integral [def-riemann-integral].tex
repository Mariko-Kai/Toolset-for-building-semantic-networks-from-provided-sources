\begin{operation}[RiemannIntegral]
\[
\TerForall \Function{f} \colon \ClosedInterval{[a,b]} \TermTo \RealNumbers
\TerForall I \colon \RealNumbers
\mathrm{IsRiemannIntegral}(\Function{f}, I) \TermDefinedAs
\TerForall \varepsilon \in \RealNumbers \left( \varepsilon > 0 \TermImplication 
\TermExists \delta \in \RealNumbers \left( \delta > 0 \And 
\TerForall n \in \NaturalNumbers \TerForall P \in \Partition{[a,b]} \TerForall \xi \in \Partition{P} 
\left( \Partition{\lambda}(P) < \delta \TermImplication 
\AbsAbstract{\left| I - \sum_{i=0}^{n-1} f(\xi(i)) \cdot (x(i+1) - x(i)) \right|} < \varepsilon \right) \right) \right)
\]
\end{operation}
