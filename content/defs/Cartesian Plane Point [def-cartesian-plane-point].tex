\hypertarget{def-cartesian-plane-point}{}
\begin{definition}[def-cartesian-plane-point]
\[
\TerForall a \colon \RealNumbers
\TerForall b \colon \RealNumbers
\mathrm{IsPoint}(a, b, p) \TermDefinedAs
\left( p = (a, b) \right) \TermConjunction
\left( p \in \CartesianProduct{\RealNumbers \times \RealNumbers} \right) \TermConjunction
\left( \exists O, X, Y \in \CartesianProduct{\RealNumbers \times \RealNumbers} \right)
\left( O = (0, 0) \right) \TermConjunction
\left( X = (a, 0) \right) \TermConjunction
\left( Y = (0, b) \right) \TermConjunction
\left( p \text{ is the vertex of the rectangle with vertices } \{O, X, Y, p\} \right)
\]
\end{definition}
