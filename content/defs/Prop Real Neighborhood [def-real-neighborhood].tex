\begin{property}[Neighborhood]
\[
\TerForall p \colon \RealNumbers
\TerForall N \subseteq \RealNumbers
\mathrm{IsNeighborhood}(N, p) \TermDefinedAs \TermExists \varepsilon \in \RealNumbers \left( \varepsilon > 0 \TermConjunction N = \{ x \colon \RealNumbers \mid p - \varepsilon < x < p + \varepsilon \} \right)
\]
\end{property}
