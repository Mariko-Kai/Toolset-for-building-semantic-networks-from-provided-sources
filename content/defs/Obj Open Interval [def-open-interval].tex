\begin{object}[Open Interval]
\[
\TerForall a, b \in \RealNumbers
\TerForall S \subseteq \RealNumbers
\mathrm{IsOpenInterval}(S, a, b) \TermDefinedAs \left( a < b 
\TermConjunction{\land} S = \{x \in \RealNumbers \TermMid{\mid} a < x < b\} \right)
\]
\end{object}
