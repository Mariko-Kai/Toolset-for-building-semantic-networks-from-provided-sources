\section{obj-set}\label{entity:obj-set}
\textbf{Тип:} object\quad \textbf{Источник:} Зорич В.А., Математический анализ, Том I (zorich-1, p. 7)

\textbf{Каноническая запись:}

\begin{object}[obj-set] % Множество (Set)


\textbf{Естественный язык:}
Множество — базовое, неопределяемое напрямую понятие математики. Множество представляет собой совокупность объектов произвольной природы, называемых его элементами. Все свойства множеств строго выводятся из аксиом системы Цермело-Френкеля (ZFC).

\section{axm-zfc-extensionality}\label{entity:axm-zfc-extensionality}
\textbf{Тип:} axiom\quad \textbf{Источник:} vereshchagin-shen (vereshchagin-shen)

\textbf{Каноническая запись:}

\begin{axiom}[axm-zfc-extensionality] % Аксиома объемности (Axiom of Extensionality)



\[
\mForall{A, B}
(A = B) \mIff (\mForall{x} x \in A \mIff x \in B)
\]
\end{axiom}

\section{term-log-conn}\label{entity:term-log-conn}
\textbf{Тип:} terminal\quad \textbf{Источник:} Unknown (Unknown)

\textbf{Каноническая запись:}

\begin{object}[term-log-conn] % Логические связки (Logical Connectives)



\textbf{Базовый набор:}
\begin{itemize}
    \item \textbf{Конъюнкция ($\entityref{term-log-conn}{\land}$):} Истинна, когда оба утверждения истинны.
    \item \textbf{Дизъюнкция ($\entityref{term-log-conn}{\lor}$):} Истинна, когда хотя бы одно утверждение истинно.
    \item \textbf{Отрицание ($\lnot$):} Инвертирует логическое значение.
    \item \textbf{Импликация ($\entityref{term-log-conn}{\implies}$):} «Если... то...».
    \item \textbf{Эквивалентность ($\entityref{term-log-conn}{\iff}$):} «Тогда и только тогда, когда».
\end{itemize}
\end{object}
